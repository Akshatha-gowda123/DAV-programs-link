    "import pandas as pd\n",
    "df=pd.read_csv('students.csv')\n",
    "Q1=df['StudyHours'].quantile(0.25)\n",
    "Q3=df['StudyHours'].quantile(0.75)\n",
    "IQR=Q3-Q1\n",
    "lower=Q1-1.5*IQR\n",
    "upper=Q3+1.5*IQR\n",
    "outlier_iqr=df[\n",
    "    (df['StudyHours']<lower)|\n",
    "    (df['StudyHours']>upper)\n",
    "    ]\n",
    "print(f'IQR bounds : [{lower:.2f}, {upper:.2f}]')\n",
    "print('Outliers found (IQR):', len(outlier_iqr))\n",
    "#Z score\n",
    "df['Z_Score']=(\n",
    "    df['StudyHours']-df['StudyHours'].mean()\n",
    ")/df['StudyHours'].std()\n",
    "outliers_z=df[df['Z_Score'].abs()>3]\n",
    "print('Outliers found (Z-Score>3):',len(outliers_z))\n",
    "#Capping\n",
    "df['StudyHours_Capped']=df['StudyHours'].clip(\n",
    "    lower = lower,\n",
    "    upper = upper\n",
    ")\n",
    "print('\\nBefore vs After capping (max value):')\n",
    "print('Before:',df['StudyHours'].max())\n",
    "print('After :',df['StudyHours_Capped'].max())\n",
    "\n",
